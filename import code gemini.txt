                <span className="flex items-center gap-1"><MapPin size={12}/> {job.location}</span>
                {job.price && <span className="flex items-center gap-1 font-bold text-slate-700"><Tag size={12}/> {job.price}</span>}
            </div>
        </div>
    );

    return (
        <div className="min-h-screen bg-slate-50 pb-20">
            {/* Modal Intervention */}
            {selectedJob && (
                <InterventionSheet 
                    job={selectedJob} 
                    userRole="artisan" 
                    onClose={() => setSelectedJob(null)} 
                    onUpdateJob={onUpdateJob}
                    suppliers={suppliers}
                    onNotify={onNotify}
                />

            {/* Notifications */}
            {showNotifications && <NotificationPanel notifications={myNotifications} onClose={() => setShowNotifications(false)} />}

            {/* HEADER */}
            <div className="bg-slate-900 text-white pt-12 pb-24 px-6 rounded-b-[2.5rem] shadow-xl relative overflow-hidden">
                <div className="flex justify-between items-center relative z-10">
                    <div>
                        <p className="text-slate-400 text-xs mb-1 uppercase tracking-wider">Espace Pro</p>
                        <h1 className="text-2xl font-bold">{artisan.prenoms} {artisan.nom}</h1>
                        <div className="flex items-center gap-2 mt-1">
                            <span className="bg-slate-800 px-2 py-0.5 rounded text-[10px] text-slate-300 border border-slate-700 capitalize">{artisan.category}</span>
                            <div className="flex items-center text-yellow-400 text-xs font-bold"><Star size={10} fill="currentColor"/> {artisan.rating}</div>
                        </div>
                    </div>
                    <button onClick={() => setShowNotifications(!showNotifications)} className="w-10 h-10 bg-slate-800 rounded-full flex items-center justify-center border border-slate-700 relative">
                        <Bell size={20} className="text-slate-300"/>
                        {myNotifications.length > 0 && <span className="absolute top-0 right-0 w-3 h-3 bg-red-500 rounded-full border-2 border-slate-900"></span>}
                    </button>
                </div>
            </div>

            {/* CONTENU PRINCIPAL - Décalé vers le haut */}
            <div className="px-6 -mt-16 relative z-20">
                {currentTab === 'dashboard' && (
                    <div className="animate-in slide-in-from-bottom-5">
                        {/* KPI GRID */}
                        <div className="grid grid-cols-2 gap-3 mb-6">
                            <StatCard title="Revenus" value={`${(totalRevenue / 1000).toFixed(1)}k`} subtext="FCFA" icon={Wallet} color="bg-green-100" />
                            <StatCard title="Succès" value={`${successRate}%`} subtext="Conversion" icon={TrendingUp} color="bg-blue-100" />
                            <StatCard title="Actifs" value={activeJobs.length} subtext="Chantiers" icon={Hammer} color="bg-orange-100" />
                            <StatCard title="Vues" value="124" subtext="+12% sem." icon={Users} color="bg-purple-100" />
                        </div>

                        {/* SECTION NOUVELLES DEMANDES (ALERTE) */}
                        {pendingRequests.length > 0 && (
                            <div className="bg-white p-4 rounded-xl border-l-4 border-red-500 shadow-sm mb-6">
                                <div className="flex justify-between items-center mb-2">
                                    <h3 className="font-bold text-red-600 flex items-center gap-2"><AlertTriangle size={18}/> Action Requise</h3>
                                    <span className="bg-red-100 text-red-700 text-xs font-bold px-2 py-1 rounded-full">{pendingRequests.length}</span>
                                </div>
                                <p className="text-xs text-gray-500 mb-3">Vous avez de nouvelles demandes de devis en attente.</p>
                                <Button onClick={() => { setCurrentTab('requests'); setFilterStatus('incoming'); }} className="w-full py-2 text-xs bg-red-50 text-red-600 border border-red-100 hover:bg-red-100">Voir les demandes</Button>
                            </div>
                        )}

                        {/* SECTION AGENDA RAPIDE */}
                        <div className="bg-white p-5 rounded-xl shadow-sm border border-gray-100">
                            <div className="flex justify-between items-center mb-4">
                                <h3 className="font-bold text-slate-800 flex items-center gap-2"><Calendar size={18}/> Planning Aujourd'hui</h3>
                            </div>
                            {activeJobs.length > 0 ? (
                                <div className="space-y-3">
                                    {activeJobs.slice(0, 2).map(job => (
                                        <div key={job.id} className="flex gap-3 border-l-2 border-blue-500 pl-3 py-1">
                                            <div className="text-xs font-bold text-gray-400 w-10">09:00</div>
                                            <div>
                                                <p className="text-sm font-bold text-slate-800">{job.clientName}</p>
                                                <p className="text-xs text-gray-500 truncate w-40">{job.location}</p>
                                            </div>
                                        </div>
                                    ))}
                                </div>
                            ) : (
                                <p className="text-center text-xs text-gray-400 py-4">Aucun chantier prévu aujourd'hui.</p>
                            )}
                        </div>
                    </div>
                )}

                {currentTab === 'requests' && (
                    <div className="animate-in slide-in-from-bottom-5 min-h-[50vh]">
                        {/* ONGLETS FILTRES */}
                        <div className="bg-white p-1 rounded-xl shadow-sm border border-gray-100 flex mb-4">
                            <button onClick={() => setFilterStatus('incoming')} className={`flex-1 py-2 text-xs font-bold rounded-lg transition-all relative ${filterStatus === 'incoming' ? 'bg-slate-800 text-white' : 'text-gray-500'}`}>
                                À traiter
                                {pendingRequests.length > 0 && <span className="absolute top-1 right-2 w-2 h-2 bg-red-500 rounded-full"></span>}
                            </button>
                            <button onClick={() => setFilterStatus('active')} className={`flex-1 py-2 text-xs font-bold rounded-lg transition-all ${filterStatus === 'active' ? 'bg-slate-800 text-white' : 'text-gray-500'}`}>En cours</button>
                            <button onClick={() => setFilterStatus('completed')} className={`flex-1 py-2 text-xs font-bold rounded-lg transition-all ${filterStatus === 'completed' ? 'bg-slate-800 text-white' : 'text-gray-500'}`}>Terminés</button>
                        </div>

                        {/* LISTE DES JOBS */}
                        <div className="space-y-2">
                            {filterStatus === 'incoming' && pendingRequests.map(job => <JobCard key={job.id} job={job}/>)}
                            {filterStatus === 'active' && activeJobs.map(job => <JobCard key={job.id} job={job}/>)}
                            {filterStatus === 'completed' && completedJobs.map(job => <JobCard key={job.id} job={job}/>)}
                            
                            {/* Empty States */}
                            {filterStatus === 'incoming' && pendingRequests.length === 0 && <div className="text-center py-10 text-gray-400 text-sm">Aucune nouvelle demande.</div>}
                            {filterStatus === 'active' && activeJobs.length === 0 && <div className="text-center py-10 text-gray-400 text-sm">Aucun chantier en cours.</div>}
                        </div>
                    </div>
                )}

                {currentTab === 'profile' && (
                    <div className="animate-in slide-in-from-bottom-5">
                         {isEditing ? (
                            <form onSubmit={handleSaveProfile} className="bg-white p-4 rounded-xl shadow-sm border border-gray-100 space-y-4">
                                <div className="flex justify-center mb-4">
                                    <div className="w-24 h-24 bg-slate-100 rounded-full border-4 border-white shadow-lg overflow-hidden">
                                        {editFormData.photo ? <img src={editFormData.photo} className="w-full h-full object-cover"/> : <User className="w-12 h-12 m-auto mt-6 text-slate-300"/>}
                                    </div>
                                </div>
                                <InputGroup label="Nom"><input name="nom" value={editFormData.nom} onChange={handleEditChange} className="w-full bg-gray-50 border rounded-xl p-3" /></InputGroup>
                                <InputGroup label="Prénoms"><input name="prenoms" value={editFormData.prenoms} onChange={handleEditChange} className="w-full bg-gray-50 border rounded-xl p-3" /></InputGroup>
                                <InputGroup label="Tarif Horaire"><input name="price" value={editFormData.price} onChange={handleEditChange} className="w-full bg-gray-50 border rounded-xl p-3" /></InputGroup>
                                <div className="bg-blue-50 p-4 rounded-xl border border-blue-100">
                                    <div className="flex gap-2 mb-3">
                                        <input readOnly value={editFormData.gpsCoords || ''} placeholder="GPS actuel" className="flex-1 bg-white border border-blue-200 rounded-lg p-3 text-xs text-gray-600" />
                                        <button type="button" onClick={handleGeo} disabled={locLoading} className="bg-blue-600 text-white px-3 py-2 rounded-lg text-xs font-bold flex items-center gap-1">{locLoading ? <Loader2 size={12} className="animate-spin"/> : <Navigation size={12}/>} GPS</button>
                                    </div>
                                    <input name="adresseManuelle" value={editFormData.adresseManuelle} onChange={handleEditChange} className="w-full bg-white border border-blue-200 rounded-lg p-3 text-sm" />
                                </div>
                                <div className="flex gap-2 pt-2"><Button variant="secondary" onClick={() => setIsEditing(false)} className="flex-1">Annuler</Button><Button type="submit" className="flex-1" icon={Save}>Enregistrer</Button></div>
                            </form>
                         ) : (
                            <div className="space-y-4">
                                <div className="bg-white p-6 rounded-2xl shadow-sm border border-gray-100 relative">
                                    <button onClick={() => { setEditFormData(artisan); setIsEditing(true); }} className="absolute top-4 right-4 p-2 bg-gray-50 rounded-full hover:bg-slate-50 text-gray-400 hover:text-slate-600"><Edit2 size={18}/></button>
                                    <div className="flex items-center gap-4 mb-4">
                                        <div className="w-16 h-16 bg-slate-100 rounded-full overflow-hidden border border-gray-100">{artisan.image ? <img src={artisan.image} className="w-full h-full object-cover"/> : <User size={32} className="m-auto mt-4 text-gray-300"/>}</div>
                                        <div>
                                            <h3 className="font-bold text-lg text-slate-800">{artisan.prenoms} {artisan.nom}</h3>
                                            <p className="text-slate-500 text-sm capitalize">{artisan.category}</p>
                                        </div>
                                    </div>
                                    <div className="space-y-2 text-sm text-gray-600">
                                        <div className="flex items-center gap-2"><Phone size={16}/> {artisan.telephone}</div>
                                        <div className="flex items-center gap-2"><MapPin size={16}/> {artisan.adresseManuelle || artisan.location}</div>
                                        <div className="flex items-center gap-2"><Tag size={16}/> {artisan.price}</div>
                                    </div>
                                </div>
                                
                                <div className="bg-white p-4 rounded-xl shadow-sm border border-gray-100">
                                    <h4 className="font-bold text-slate-800 mb-3 text-sm uppercase">Badges & Certifications</h4>
                                    <div className="flex flex-wrap gap-2">
                                        {artisan.isAffiliated && <span className="bg-blue-100 text-blue-700 px-3 py-1 rounded-full text-xs font-bold flex items-center gap-1"><ShieldCheck size={12}/> Certifié CNMCI</span>}
                                        {(artisan.badges || []).map(b => <span key={b} className="bg-gray-100 text-gray-600 px-3 py-1 rounded-full text-xs font-bold">{b}</span>)}
                                    </div>
                                </div>

                                <Button variant="danger" className="w-full" onClick={onLogout} icon={LogOut}>Déconnexion</Button>
                            </div>
                         )}
                    </div>
                )}
            </div>

            {/* NAVIGATION BOTTOM */}
            <div className="fixed bottom-0 w-full max-w-md bg-white border-t py-3 px-6 flex justify-between z-50">
                <button onClick={() => setCurrentTab('dashboard')} className={currentTab === 'dashboard' ? 'text-slate-800' : 'text-gray-400 flex flex-col items-center gap-1 text-[10px]'}>
                    <Home size={20}/><span>Accueil</span>
                </button>
                <button onClick={() => setCurrentTab('requests')} className={currentTab === 'requests' ? 'text-slate-800 relative' : 'text-gray-400 flex flex-col items-center gap-1 text-[10px] relative'}>
                    <ClipboardList size={20}/>
                    <span>Chantiers</span>
                    {pendingRequests.length > 0 && <span className="absolute -top-1 right-2 w-2.5 h-2.5 bg-red-500 rounded-full border border-white"></span>}
                </button>
                <button onClick={() => setCurrentTab('profile')} className={currentTab === 'profile' ? 'text-slate-800' : 'text-gray-400 flex flex-col items-center gap-1 text-[10px]'}>
                    <User size={20}/><span>Profil</span>
                </button>
            </div>
        </div>
    );
};

// --- UNIVERS CLIENT ---
const ClientAppFlow = ({ user, onUpdateUser, onLogout, onAddJob, jobs, onUpdateJob, artisans, suppliers, orders, onAddOrder, onUpdateOrder, notifications = [], onNotify }) => { // <--- Receive notifications and onNotify
  const [currentTab, setCurrentTab] = useState('home');
  const [selectedArtisan, setSelectedArtisan] = useState(null);
  const [showAI, setShowAI] = useState(false);
  const [categoryFilter, setCategoryFilter] = useState(null);
  const [problemDescription, setProblemDescription] = useState("");
  const [sendingQuote, setSendingQuote] = useState(false);
  const [showSuccessQuote, setShowSuccessQuote] = useState(false);
  const [showNeedsInput, setShowNeedsInput] = useState(false);
  const [selectedJobForSheet, setSelectedJobForSheet] = useState(null);
  const [isEditingProfile, setIsEditingProfile] = useState(false);
  const [editFormData, setEditFormData] = useState({});
  const [locLoading, setLocLoading] = useState(false);
  const [searchType, setSearchType] = useState('artisan'); 
  // New State for Cart & Notifications
  const [cart, setCart] = useState([]);
  const [showCart, setShowCart] = useState(false);
  const [showRating, setShowRating] = useState(false);
  const [scanResult, setScanResult] = useState(null);
  const [scanCode, setScanCode] = useState('');
  const [activeOrderForScan, setActiveOrderForScan] = useState(null);
  const [showNotifications, setShowNotifications] = useState(false);

  useEffect(() => { if (user) setEditFormData(user); }, [user]);
  const handleEditChange = (e) => setEditFormData({ ...editFormData, [e.target.name]: e.target.value });
  const handleGeo = () => { setLocLoading(true); navigator.geolocation.getCurrentPosition((pos) => { setEditFormData(prev => ({ ...prev, gpsCoords: `${pos.coords.latitude.toFixed(6)}, ${pos.coords.longitude.toFixed(6)}` })); setLocLoading(false); }, () => { alert("Erreur GPS"); setLocLoading(false); }); };
  const handleSaveProfile = (e) => { e.preventDefault(); onUpdateUser(editFormData); setIsEditingProfile(false); };

  const handleSendQuoteRequest = () => {
    if (!problemDescription.trim()) { alert("Veuillez décrire votre besoin."); return; }
    setSendingQuote(true);
    setTimeout(() => {
        const newJob = {
            id: Date.now(), clientName: `${user.nom} ${user.prenoms}`, clientId: user.id, clientPhone: user.telephone, clientAddress: user.adresseManuelle, location: user.adresseManuelle || "Abidjan", problem: problemDescription, date: new Date().toLocaleDateString(), price: null, status: 'sent', artisanId: selectedArtisan.id, artisanName: `${selectedArtisan.name} ${selectedArtisan.prenoms}`, artisanPhone: selectedArtisan.telephone || "00000000", artisanCategory: selectedArtisan.category, expiresAt: Date.now() + (48 * 60 * 60 * 1000), interventionData: null,
            paymentStatus: 'pending', financials: { totalMaterials: 0, totalLabor: 0, tokenCode: null, tokenAmount: 0, tokenUsed: 0, escrowLabor: 0, escrowMaterials: 0 }, milestones: [{ id: 1, name: "Livraison Matériel", status: 'pending', proof: null, amountRelease: 0 }, { id: 2, name: "Fin des Travaux", status: 'pending', proof: null, amountRelease: 0 }]
        };
        onAddJob(newJob);
        // Trigger Notification for Artisan
        onNotify(selectedArtisan.id, 'artisan', `Nouvelle demande de devis de ${user.nom} ${user.prenoms}.`);
        
        setSendingQuote(false);
        setShowSuccessQuote(true);
        setTimeout(() => {
            setShowSuccessQuote(false);
            setShowNeedsInput(false);
            setProblemDescription("");
            setSelectedArtisan(null);
            setCurrentTab('requests');
        }, 2000);
    }, 1500);
  };

  const handleAddToCart = (item, supplierName, supplierId) => {
      if (cart.length > 0 && cart[0].supplierId !== supplierId) {
          if (!confirm("Voulez-vous vider le panier pour ajouter des articles d'un autre fournisseur ?")) return;
          setCart([]);
      }
      setCart(prev => [...prev, { ...item, supplierName, supplierId, quantity: 1 }]);
      alert("Article ajouté au panier !");
  };

  const handleRemoveFromCart = (index) => {
      const newCart = [...cart];
      newCart.splice(index, 1);
      setCart(newCart);
  };

  const handleSubmitOrder = (order) => {
      onAddOrder({ ...order, clientId: user.id || 999 });
      // Notify Supplier
      onNotify(order.supplierId, 'supplier', `Nouvelle commande de ${user.nom} ${user.prenoms} (${order.total} F).`);
      setCart([]);
      setShowCart(false);
      alert("Commande validée avec succès !");
  };

  const handleClientScanDelivery = () => {
      if(scanCode === 'LIVREUR-CODE-CONFIRM' || scanCode.length > 5) {
          onUpdateOrder(activeOrderForScan.id, { status: 'delivered' });
          setActiveOrderForScan(null);
          setScanCode('');
          setShowRating(true); 
      } else {
          alert("Code invalide (Essayez LIVREUR-CODE-CONFIRM)");
      }
  };

  const myRequests = jobs.filter(j => j.clientName === `${user.nom} ${user.prenoms}`);
  const myOrders = orders;
  const myNotifications = notifications.filter(n => n.role === 'client' && (!n.userId || n.userId === user.id));

  return (
    <div className="bg-gray-50 min-h-screen">
      <AIAssistantModal isOpen={showAI} onClose={() => setShowAI(false)} onCategoryFound={(cat) => { setShowAI(false); setCategoryFilter(cat); setCurrentTab('search'); }} />
      {/* Pass onNotify to InterventionSheet */}
      {selectedJobForSheet && (<InterventionSheet job={selectedJobForSheet} userRole="client" onClose={() => setSelectedJobForSheet(null)} onUpdateJob={onUpdateJob} suppliers={suppliers} onNotify={onNotify} />)}
      
      {showRating && <RatingModal onClose={() => setShowRating(false)} />}
      
      {/* NOTIFICATIONS PANEL */}
      {showNotifications && <NotificationPanel notifications={myNotifications} onClose={() => setShowNotifications(false)} />}

      {/* SCANNER CLIENT (POUR LIVRAISON) */}
      {activeOrderForScan && (
          <div className="fixed inset-0 bg-black/80 z-[70] flex items-center justify-center p-6 animate-in zoom-in-95">
              <div className="bg-white rounded-3xl p-6 w-full max-w-sm text-center">
                  <h3 className="text-xl font-bold mb-4">Confirmer Réception</h3>
                  <p className="text-gray-500 mb-4">Scannez le code présenté par le livreur.</p>
                  <input type="text" placeholder="Code Livreur" className="w-full border p-3 rounded-xl mb-4 text-center" value={scanCode} onChange={e => setScanCode(e.target.value)} />
                  <div className="flex gap-2">
                      <Button variant="ghost" className="flex-1" onClick={() => setActiveOrderForScan(null)}>Annuler</Button>
                      <Button variant="success" className="flex-1" onClick={handleClientScanDelivery}>Valider</Button>
                  </div>
              </div>
          </div>
      )}

      {/* MODALE PANIER */}
      {showCart && (
          <CartModal 
            cart={cart} 
            onRemoveItem={handleRemoveFromCart} 
            onClose={() => setShowCart(false)}
            onSubmitOrder={handleSubmitOrder}
          />
      )}

      {/* Floating Cart Button */}
      {cart.length > 0 && !showCart && (
          <button 
            onClick={() => setShowCart(true)}
            className="fixed bottom-20 right-6 bg-blue-600 text-white p-4 rounded-full shadow-lg z-40 animate-bounce"
          >
              <div className="relative">
                  <ShoppingCart size={24}/>
                  <span className="absolute -top-3 -right-3 bg-red-500 text-white text-xs font-bold w-5 h-5 flex items-center justify-center rounded-full">{cart.length}</span>
              </div>
          </button>
      )}

      {selectedArtisan ? (
          <div className="bg-white min-h-screen pt-12 px-6 flex flex-col"><button onClick={() => { setSelectedArtisan(null); setShowNeedsInput(false); }} className="mb-4"><ChevronLeft/></button>
              {showSuccessQuote ? (<div className="flex-1 flex flex-col items-center justify-center animate-in fade-in zoom-in"><div className="w-20 h-20 bg-green-100 rounded-full flex items-center justify-center text-green-600 mb-6"><CheckCircle size={40}/></div><h2 className="text-2xl font-bold text-center">Demande Envoyée !</h2><p className="text-center text-gray-500 mt-2">L'artisan a reçu votre demande de devis.</p></div>) : (<><div className="relative mb-6"><img src={selectedArtisan.image || "https://images.unsplash.com/photo-1581578731117-104f2a863ecc?auto=format&fit=crop&q=80&w=200"} className="w-full h-48 object-cover rounded-2xl bg-gray-200 shadow-lg" /><div className="absolute -bottom-6 right-6 w-12 h-12 bg-white rounded-xl shadow-md flex items-center justify-center font-bold text-lg text-blue-600 border border-gray-100">{selectedArtisan.completionRate || 95}<span className="text-[8px]">%</span></div></div><div className="mb-6"><h1 className="text-2xl font-bold text-gray-800">{selectedArtisan.name} {selectedArtisan.prenoms}</h1><p className="text-gray-500 text-sm mb-4 leading-relaxed">{selectedArtisan.description || "Professionnel inscrit sur la plateforme."}</p></div><div className="mt-auto pb-8 space-y-4">{!showNeedsInput ? (<Button variant="outline" className="w-full" icon={FileText} onClick={() => setShowNeedsInput(true)}>Description Besoin</Button>) : (<div className="animate-in slide-in-from-bottom-5 fade-in"><label className="text-xs font-bold uppercase text-gray-500 mb-2 block">Détails de votre problème</label><textarea className="w-full h-32 bg-gray-50 border border-blue-200 rounded-xl p-4 text-sm outline-none mb-4" placeholder="Décrivez ici..." value={problemDescription} onChange={(e) => setProblemDescription(e.target.value)} autoFocus /><div className="flex gap-2"><Button variant="ghost" onClick={() => setShowNeedsInput(false)}>Annuler</Button><Button className="flex-1" icon={sendingQuote ? Loader2 : Send} onClick={handleSendQuoteRequest} disabled={sendingQuote}>{sendingQuote ? 'Envoi...' : 'Demande de Devis'}</Button></div></div>)}</div></>)}</div>
      ) : (
        <>{currentTab === 'home' && (<div className="space-y-6 pb-24"><div className="bg-blue-600 pt-12 pb-8 px-6 rounded-b-3xl shadow-lg text-white"><div className="flex justify-between items-center mb-6"><div><p className="text-blue-100 text-sm">Bonjour, {user.prenoms}</p><h1 className="text-2xl font-bold">Besoin d'un pro ?</h1></div>
        {/* BELL ICON FOR NOTIFICATIONS */}
        <div className="flex gap-3">
             <button onClick={() => setShowNotifications(!showNotifications)} className="w-10 h-10 bg-blue-500 rounded-full flex items-center justify-center font-bold border-2 border-blue-300 relative">
                 <Bell size={20}/>
                 {myNotifications.length > 0 && <span className="absolute top-0 right-0 w-3 h-3 bg-red-500 rounded-full border-2 border-blue-500"></span>}
             </button>
             <div className="w-10 h-10 bg-blue-500 rounded-full flex items-center justify-center font-bold border-2 border-blue-300">{user.nom[0]}</div>

// --- COMPOSANT PRINCIPAL (App) ---
const App = () => {
    const [user, setUser] = useState(null);
    const [screen, setScreen] = useState('role_selection');
    const [jobs, setJobs] = useState(ARTISAN_INCOMING_JOBS);
    const [orders, setOrders] = useState([]); // Pour le e-commerce
    const [suppliers, setSuppliers] = useState(SUPPLIERS);
    const [artisans, setArtisans] = useState(ARTISANS);
    const [notifications, setNotifications] = useState([]); // Centralized Notifications

    const handleSelectRole = (role) => {
        if (role === 'client') setScreen('register_client');
        else if (role === 'artisan') setScreen('register_artisan');
        else if (role === 'supplier') setScreen('register_supplier');
    };

    const handleRegisterClient = (data) => {
        const newUser = { id: Date.now(), role: 'client', ...data };
        setUser(newUser);
        setScreen('app');
    };

    const handleRegisterArtisan = (data) => {
        const newUser = { id: Date.now(), role: 'artisan', ...data };
        setUser(newUser);
        setArtisans([...artisans, newUser]);
        setScreen('app');
    };

    const handleRegisterSupplier = (data) => {
        const newUser = { id: Date.now(), role: 'supplier', ...data };
        setUser(newUser);
        setSuppliers([...suppliers, newUser]);
        setScreen('app');
    };

    const handleLogin = (identifier, password) => {
        // Simple mock login logic
        if (identifier === "admin") { setUser({ id: 99, role: 'client', nom: "Admin", prenoms: "Test", telephone: "0000" }); setScreen('app'); return; }
        // Try finding in suppliers
        const supplier = suppliers.find(s => (s.email === identifier || s.contactMobile === identifier) && s.password === password);
        if (supplier) { setUser(supplier); setScreen('app'); return; }
        // Try finding in artisans
        const artisan = artisans.find(a => (a.email === identifier || a.telephone === identifier) && a.password === password);
        if (artisan) { setUser(artisan); setScreen('app'); return; }
        // Try finding in clients (mocked since we don't persist clients in this demo state except user state)
        // For demo, create a generic client user if not admin
        setUser({ id: Date.now(), role: 'client', nom: "Utilisateur", prenoms: "Client", telephone: identifier, adresseManuelle: "Abidjan" });
        setScreen('app');
    };

    // Notification Helper
    const addNotification = (userId, role, message) => {
        const newNotif = { id: Date.now(), userId, role, message, date: new Date().toLocaleString() };
        setNotifications(prev => [newNotif, ...prev]);
    };

    const handleUpdateUser = (updatedUser) => {
        setUser(updatedUser);
        if (updatedUser.role === 'supplier') {
            setSuppliers(suppliers.map(s => s.id === updatedUser.id ? updatedUser : s));
        } else if (updatedUser.role === 'artisan') {
            setArtisans(artisans.map(a => a.id === updatedUser.id ? updatedUser : a));
        }
    };

    return (
        <div className="font-sans text-gray-900 bg-white max-w-md mx-auto min-h-screen shadow-2xl overflow-hidden relative">
            {screen === 'role_selection' && <RoleSelectionScreen onSelectRole={handleSelectRole} onGoToLogin={() => setScreen('login')} />}
            {screen === 'login' && <LoginScreen onLogin={handleLogin} onBack={() => setScreen('role_selection')} />}
            {screen === 'register_client' && <ClientRegistrationScreen onRegister={handleRegisterClient} onBack={() => setScreen('role_selection')} />}
            {screen === 'register_artisan' && <ArtisanRegistrationScreen onRegister={handleRegisterArtisan} onBack={() => setScreen('role_selection')} />}
            {screen === 'register_supplier' && <SupplierRegistrationScreen onRegister={handleRegisterSupplier} onBack={() => setScreen('role_selection')} />}
            
            {screen === 'app' && user && user.role === 'client' && (
                <ClientAppFlow 
                    user={user} 
                    onUpdateUser={handleUpdateUser}
                    onLogout={() => setScreen('role_selection')} 
                    onAddJob={(job) => setJobs([job, ...jobs])}
                    jobs={jobs}
                    onUpdateJob={(id, updatedJob) => setJobs(jobs.map(j => j.id === id ? updatedJob : j))}
                    artisans={artisans}
                    suppliers={suppliers}
                    orders={orders}
                    onAddOrder={(order) => setOrders([order, ...orders])}
                    onUpdateOrder={(id, updated) => setOrders(orders.map(o => o.id === id ? { ...o, ...updated } : o))}
                    notifications={notifications}
                    onNotify={addNotification}
                />
            )}

            {screen === 'app' && user && user.role === 'artisan' && (
                <ArtisanDashboard 
                    artisan={user} 
                    jobs={jobs}
                    onUpdateJob={(id, updatedJob) => setJobs(jobs.map(j => j.id === id ? updatedJob : j))}
                    onUpdateUser={handleUpdateUser}
                    onLogout={() => setScreen('role_selection')}
                    notifications={notifications}
                    onNotify={addNotification}
                    suppliers={suppliers}
                />
            )}

            {screen === 'app' && user && user.role === 'supplier' && (
                <SupplierDashboard 
                    supplier={user} 
                    onUpdateUser={handleUpdateUser} 
                    onLogout={() => setScreen('role_selection')}
                    orders={orders}
                    onUpdateOrder={(id, updated) => setOrders(orders.map(o => o.id === id ? { ...o, ...updated } : o))}
                    notifications={notifications}
                    onNotify={addNotification}
                />
            )}
        </div>
    );
};

export default App;
